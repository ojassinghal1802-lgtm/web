        function getCurrentAcademicSession() {
            const now = new Date();
            const year = now.getFullYear();
            const month = now.getMonth(); 
            let startYear = (month < 3) ? year - 1 : year;
            let endYear = startYear + 1;
            let startYY = startYear.toString().slice(-2);
            let endYY = endYear.toString().slice(-2);
            return `${startYear}-${endYear} (${startYY}-${endYY})`;
        }

        document.addEventListener('touchmove', drag, {passive: true});

        function dragStart(e) {
            isDragging = true;
            startX = e.clientX || e.touches[0].clientX;
            startY = e.clientY || e.touches[0].clientY;
            initialX = chatbotWidget.offsetLeft;
            initialY = chatbotWidget.offsetTop;
        }

        function drag(e) {
            if (!isDragging) return;
            const currentX = e.clientX || e.touches[0].clientX;
            const currentY = e.clientY || e.touches[0].clientY;
            const dx = currentX - startX;
            const dy = currentY - startY;

            chatbotWidget.style.left = (initialX + dx) + 'px';
            chatbotWidget.style.top = (initialY + dy) + 'px';
            chatbotWidget.style.bottom = 'auto';
            chatbotWidget.style.right = 'auto';
        }

        function dragEnd() {
            isDragging = false;
        }

        function verifyApplication(appId) {
            let applications = JSON.parse(localStorage.getItem('knga_confirmed_admissions') || '[]');
            applications = applications.map(app => {
                if (app.id === appId) {
                    app.status = 'Verified & Confirmed';
                }
                return app;
            });
            localStorage.setItem('knga_confirmed_admissions', JSON.stringify(applications));
            loadAdminApplications();
        }

        function loadAdminApplications() {
            const applications = JSON.parse(localStorage.getItem('knga_confirmed_admissions') || '[]');
            const tbody = document.getElementById('admin-apps-table');
            const badge = document.getElementById('app-count-badge');
            badge.innerText = applications.length + ' Applications';

            if (applications.length === 0) {
                tbody.innerHTML = `<tr><td colspan="6" class="p-6 text-center text-slate-500">No student applications submitted yet.</td></tr>`;
                return;
            }

            tbody.innerHTML = '';
            applications.forEach(app => {
                let statusActionHTML = `<button onclick="verifyApplication('${app.id}')" class="bg-amber-600 hover:bg-amber-700 text-white px-3 py-1.5 rounded font-bold text-[10px] uppercase shadow">Verify Payment</button>`;
                if (app.status === 'Verified & Confirmed') {
                    statusActionHTML = `<button onclick="viewVerifiedSlip('${app.id}')" class="bg-emerald-600 hover:bg-emerald-700 text-white px-3 py-1.5 rounded font-bold text-[10px] uppercase shadow">View & Print Slip</button>`;
                }

                tbody.innerHTML += `
                    <tr class="hover:bg-slate-50 border-b border-slate-100">
                        <td class="p-3 font-mono font-bold text-brand-900">${app.id}</td>
                        <td class="p-3 font-semibold text-slate-800">${app.name}</td>
